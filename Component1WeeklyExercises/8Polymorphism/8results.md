class Secret implements Encryptable { private String message;

public Secret(String message) {
    this.message = message;
}

public void encrypt() {
    char[] chars = message.toCharArray();
    for (int i = 0; i < chars.length; i++) {
        chars[i] += 3; // Caesar shift
    }
    message = new String(chars);
    System.out.println("Encrypted Secret: " + message);
}

public void decrypt() {
    char[] chars = message.toCharArray();
    for (int i = 0; i < chars.length; i++) {
        chars[i] -= 3;
    }
    message = new String(chars);
    System.out.println("Decrypted Secret: " + message);
}
} public class TestEncryption { public static void main(String[] args) {

    Encryptable obj;

    obj = new Secret("HelloWorld");
    obj.encrypt();
    obj.decrypt();

    obj = new Password("MyPass123");
    obj.encrypt();
    obj.decrypt();
}
} class Password implements Encryptable { private String password;

public Password(String password) {
    this.password = password;
}

public void encrypt() {
    password = new StringBuilder(password).reverse().toString();
    System.out.println("Encrypted Password: " + password);
}

public void decrypt() {
    password = new StringBuilder(password).reverse().toString();
    System.out.println("Decrypted Password: " + password);
}
} interface Encryptable { void encrypt(); void decrypt(); }
