import java.util.Scanner;

public class Histogram { public static void main(String[] args) { Scanner sc = new Scanner(System.in);

    int[] freq = new int[10]; // 10 ranges

    System.out.println("Enter numbers (1–100), -1 to stop:");

    while (true) {
        int num = sc.nextInt();

        if (num == -1)
            break;

        if (num >= 1 && num <= 100) {
            int index = (num - 1) / 10;
            freq[index]++;
        } else {
            System.out.println("Invalid number!");
        }
    }

    // Print histogram
    for (int i = 0; i < 10; i++) {
        int start = i * 10 + 1;
        int end = (i + 1) * 10;

        System.out.printf("%2d - %3d | ", start, end);

        for (int j = 0; j < freq[i]; j++) {
            System.out.print("*");
        }
        System.out.println();
    }

    sc.close();
}

                break;

                System.out.print("Account index: ");

                if (i >= 0 && i < count) { // ✅ validation added
                    System.out.print("Amount: ");
                    accounts[i].withdraw(sc.nextDouble());

} class Account { private String name; private double balance;

public Account(String name, double balance) {
    this.name = name;
    this.balance = balance;
}

public void deposit(double amount) {
    if (amount > 0)
        balance += amount;
    else
        System.out.println("Invalid deposit amount!");
}

public void withdraw(double amount) {
    if (amount > balance)
        System.out.println("Insufficient balance!");
    else if (amount <= 0)
        System.out.println("Invalid withdrawal!");
    else
        balance -= amount;
}

public void addInterest() {
    balance += balance * 0.03;
}

public void display() {
    System.out.println(name + " Balance: " + balance);
}

import java.util.Scanner;

public class PostCodes { public static void main(String[] args) { Scanner sc = new Scanner(System.in);

    Person[] list = new Person[25];
    int count = 0;

    System.out.println("Enter data (first last postal), type 'stop' to end:");

    while (count < 25) {
        String input = sc.nextLine();

        if (input.equalsIgnoreCase("stop"))
            break;

        String[] parts = input.split(" ");

        if (parts.length != 3) {
            System.out.println("Invalid input! Enter: first last postal");
            continue;
        }

        String first = parts[0];
        String last = parts[1];
        String code = parts[2];

        list[count++] = new Person(first, last, code);
    }

    System.out.println("\n--- Stored Data ---");
    for (int i = 0; i < count; i++) {
        list[i].display();
    }

    sc.close();
}
} class Person { String firstName, lastName, postalCode;

public Person(String f, String l, String p) {
    firstName = f;
    lastName = l;
    postalCode = p;
}

public void display() {
    System.out.println(firstName + " " + lastName + " - " + postalCode);
}

class Quiz { private Question[] questions = new Question[25]; private int count = 0;

public void add(Question q) {
    if (count < 25)
        questions[count++] = q;
}

public void giveQuiz() {
    java.util.Scanner sc = new java.util.Scanner(System.in);
    int score = 0;

    for (int i = 0; i < count; i++) {
        if (questions[i].ask(sc)) {
            System.out.println("Correct!");
            score++;
        } else {
            System.out.println("Wrong!");
        }
    }

    System.out.println("\nFinal Score: " + score + "/" + count);
}
} class Question { private String question; private String answer;

public Question(String q, String a) {
    question = q;
    answer = a;
}

public boolean ask(java.util.Scanner sc) {
    System.out.println(question);
    String userAnswer = sc.nextLine();

    return userAnswer.equalsIgnoreCase(answer);
}
} public class QuizTime { public static void main(String[] args) {

    Quiz quiz = new Quiz();

    quiz.add(new Question("Capital of Nepal?", "Kathmandu"));
    quiz.add(new Question("2 + 2 = ?", "4"));
    quiz.add(new Question("Java is platform independent? (yes/no)", "yes"));

    quiz.giveQuiz();
}
