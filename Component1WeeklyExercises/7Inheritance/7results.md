class PlayerStats {
 protected String playerName;
 protected String teamName;
 protected int gamesPlayed;

 public PlayerStats(String playerName, String teamName, int gamesPlayed) {
     this.playerName = playerName;
     this.teamName = teamName;
     this.gamesPlayed = gamesPlayed;
 }

 public void displayGeneralInfo() {
     System.out.println("Player: " + playerName + " | Team: " + teamName + " | Games: " + gamesPlayed);
 }
}

//Football subclass (Soccer/American Football context)
class FootballStats extends PlayerStats {
 private int goals;
 private int assists;

 public FootballStats(String name, String team, int games, int goals, int assists) {
     super(name, team, games);
     this.goals = goals;
     this.assists = assists;
 }

 public void displaySportStats() {
     double goalsPerGame = (double) goals / gamesPlayed;
     System.out.println("[Football] Goals: " + goals + ", Assists: " + assists + 
                        ", Avg Goals/Game: " + String.format("%.2f", goalsPerGame));
 }
}

class CricketStats extends PlayerStats {

 public CricketStats(String name, String team, int games, int runs, double avg) {
     super(name, team, games);
     this.runsScored = runs;
     this.battingAverage = avg;
 }

 public void displaySportStats() {
     System.out.println("[Cricket] Total Runs: " + runsScored + ", Batting Avg: " + battingAverage);
 }
}
